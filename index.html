document.getElementById('formSiswa').addEventListener('submit', async (e) => { e.preventDefault(); const btn = document.getElementById('btnSaveSiswa'); const alertEl = document.getElementById('alertSiswa'); // Ambil elemen input const elNis = document.getElementById('siswaNis') || document.querySelector('input[name="nis"]'); const elNama = document.getElementById('siswaNama') || document.querySelector('input[name="nama"]'); const elKelas = document.getElementById('siswaKelas') || document.querySelector('input[name="kelas"]'); const elRfid = document.getElementById('siswaRfid') || document.querySelector('input[name="rfid_uid"]'); const nisVal = elNis ? elNis.value.trim() : ''; const namaVal = elNama ? elNama.value.trim() : ''; const kelasVal = elKelas ? elKelas.value.trim() : ''; const rfidVal = elRfid ? elRfid.value.trim() : ''; if (!namaVal || !kelasVal) { showAlert(alertEl, 'Silakan isi Nama dan Kelas terlebih dahulu!', 'error'); return; } btn.disabled = true; btn.innerText = 'Menyimpan...'; const payload = { nis: nisVal, nama: namaVal, kelas: kelasVal, rfid_uid: rfidVal || null }; try { const res = await fetch('/api/siswa', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' }, body: JSON.stringify(payload) }); const data = await res.json(); if (res.ok && data.success) { closeModalSiswa(); loadSiswa(); } else { showAlert(alertEl, data.message || 'Gagal menyimpan data siswa.', 'error'); } } catch (err) { showAlert(alertEl, `Terjadi kesalahan: ${err.message}`, 'error'); } finally { btn.disabled = false; btn.innerText = 'Simpan Data'; } });
